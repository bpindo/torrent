Mini Fullstack GAS + Drive Learning

Ini adalah panduan belajar membuat **web app sederhana** dengan:

  • Frontend HTML + JavaScript
  • Backend Google Apps Script (GAS)
  • Database sederhana di Google Drive (JSON shard)

1️⃣ Langkah Persiapan Google Apps Script

Buka Google Drive → New → More → Google Apps Script

  1. Buat project baru, misal: MiniUserDB
  2. Buat file Code.gs dengan isi backend (lihat kode di bawah)
  3. Deploy sebagai Web App: Deploy → New deployment → Web App → Execute as: Me → Who has access: Anyone
  4. Catat URL Web App, nanti dipakai di frontend
// ========================
// CONFIG
// ========================
const SHARD_SIZE = 1000;
const FOLDER_NAME = "GAS_UserDB";

// ========================
// FOLDER & DRIVE HELPERS
// ========================
function getFolder(){ 
  const f = DriveApp.getFoldersByName(FOLDER_NAME);
  return f.hasNext()? f.next(): DriveApp.createFolder(FOLDER_NAME);
}

function getShardFile(shardKey){
  const files = getFolder().getFilesByName(shardKey+".json");
  return files.hasNext()? files.next(): null;
}

function saveShard(shardKey,data){
  const json = JSON.stringify(data);
  const file = getShardFile(shardKey);
  if(file) file.setContent(json);
  else getFolder().createFile(shardKey+".json",json,MimeType.JSON);
}

function loadShard(shardKey){
  const file = getShardFile(shardKey);
  return file ? JSON.parse(file.getBlob().getDataAsString()) : {};
}

// ========================
// SHARDING
// ========================
function getShardKey(userId){ return "users_"+(Math.floor(userId/SHARD_SIZE)+1); }

// ========================
// CRUD USER
// ========================
function saveUser(userId,data){
  const shard = loadShard(getShardKey(userId));
  shard[userId]=data;
  saveShard(getShardKey(userId),shard);
}

function getUser(userId){
  const shard = loadShard(getShardKey(userId));
  return shard[userId] || null;
}

// ========================
// USER MANAGEMENT
// ========================
function createUser(userId,username){
  const now = new Date().getTime();
  const user = {id:userId, username, posting:[], createdAt:now, lastActive:now};
  saveUser(userId,user);
}

function addPosting(userId,konten){
  const user = getUser(userId);
  if(!user) return;
  const now = new Date().getTime();
  const postId = user.posting.length ? user.posting[user.posting.length-1].id+1 :1;
  user.posting.push({id:postId,konten,createdAt:now});
  user.lastActive = now;
  saveUser(userId,user);
}

// ========================
// LIST USERS
// ========================
function listUsers(){
  const folder = getFolder();
  const files = folder.getFiles();
  const users = [];
  while(files.hasNext()){
    const file = files.next();
    if(file.getName().endsWith(".json")){
      const shard = JSON.parse(file.getBlob().getDataAsString());
      for(let uid in shard){
        users.push(shard[uid]);
      }
    }
  }
  return users;
}

// ========================
// WEB APP API
// ========================
function doGet(){ return ContentService.createTextOutput("Use POST with JSON payload"); }

function doPost(e){
  const params = JSON.parse(e.postData.contents);
  let res={status:"error",message:"Unknown action"};
  switch(params.action){
    case "createUser": createUser(params.userId,params.username); res={status:"ok"}; break;
    case "addPosting": addPosting(params.userId,params.konten); res={status:"ok"}; break;
    case "getUser": res={status:"ok",data:getUser(params.userId)}; break;
    case "listUsers": res={status:"ok",data:listUsers()}; break;
  }
  return ContentService.createTextOutput(JSON.stringify(res))
        .setMimeType(ContentService.MimeType.JSON);
}

<script>
const API_URL = "PASTE_DEPLOYED_WEB_APP_URL_HERE";

function apiPost(payload, cb){
  fetch(API_URL,{method:"POST",body:JSON.stringify(payload)})
    .then(r=>r.json()).then(cb).catch(err=>document.getElementById("output").innerText=err);
}

function createUser(){
  const userId = Number(document.getElementById("userId").value);
  const username = document.getElementById("username").value;
  apiPost({action:"createUser", userId, username}, r=>{
    document.getElementById("output").innerText=JSON.stringify(r,null,2);
  });
}

function addPosting(){
  const userId = Number(document.getElementById("userId").value);
  const konten = document.getElementById("konten").value;
  apiPost({action:"addPosting", userId, konten}, r=>{
    document.getElementById("output").innerText=JSON.stringify(r,null,2);
  });
}

function getUser(){
  const userId = Number(document.getElementById("getUserId").value);
  apiPost({action:"getUser", userId}, r=>{
    document.getElementById("output").innerText=JSON.stringify(r,null,2);
  });
}

function listUsers(){
  apiPost({action:"listUsers"}, r=>{
    document.getElementById("output").innerText=JSON.stringify(r,null,2);
    renderUserTable(r.data);
  });
}

function renderUserTable(users){
  if(!users || !users.length){document.getElementById("userTable").innerHTML="No users";return;}
  let html = "<table><tr><th>ID</th><th>Username</th><th>Posting</th></tr>";
  users.forEach(u=>{
    html+="<tr><td>"+u.id+"</td><td>"+u.username+"</td><td>"+(u.posting?u.posting.map(p=>p.konten).join(", "):"")+"</td></tr>";
  });
  html+="</table>";
  document.getElementById("userTable").innerHTML=html;
}
</script>

3️⃣ Cara Menggunakan

  1. Buka HTML di browser
  2. Masukkan User ID & Username → klik "Create User"
  3. Tambah posting → klik "Add Posting"
  4. Lihat user → masukkan User ID → klik "Get User"
  5. Lihat semua user → klik "List Users"
  6. Semua data tersimpan di Google Drive secara shard JSON → belajar konsep database + sharding sederhana

4️⃣ Catatan Pembelajaran

  • Backend GAS berfungsi sebagai API sederhana
  • Drive menyimpan data JSON → database mini
  • Sharding membantu mengatur data ribuan user
  • Frontend HTML berkomunikasi via fetch POST → belajar konsep fullstack
  • Bisa dikembangkan: hapus user, filter posting, real-time update dengan setInterval
